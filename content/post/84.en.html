This fingerprint lock is installed on our dorm room door. A 14 g metal-geared servo operates the lock—yes, our university's door locks really are that bad…
The board is an Arduino Mega, connected to a servo, a 4×4 keypad (the key layout is in the code), a 5110LCD, a microswitch (which switches on power to the fingerprint module), and the star of the show: the TFS-M74A fingerprint module.
By the way, Shizhi Technology's fingerprint modules use compatible protocols and are interchangeable. Their maximum storage capacities differ, so you'll need to change that setting (find it yourself, young grasshopper!).
The connections and such are shown in the code. I should have written them down, but I was too lazy… The 5110 library is LCD5110_Graph; search Baidu if you don't have it.
I live in an eight-person dorm room (those of you in four-person rooms or smaller, please don't rub it in…), so I reserved 16 entries in the names array—one finger from each hand per person. You can change that.

  82. uint8_t TxAndRsCmd(uint8_t packet[], uint8_t cmd, uint8_t p1 = 0, uint8_t p2 = 0, uint8_t p3 = 0, uint8_t p4 = 0)   
  83. {   
  84.   // 发送段   
  85.   uint8_t chk = 0;   
  86.      
  87.   chk ^= cmd;

  91.   chk ^= p4;   
  92.   
  93.   Serial1.write((uint8_t)CMD_HEAD);   
  94.   Serial1.write((uint8_t)cmd);   
  95.   Serial1.write((uint8_t)p1);   
  96.   Serial1.write((uint8_t)p2);

  98.   Serial1.write((uint8_t)p4);

  101.   
  102.   // 接收段   
  103.   int timer = 0;   
  104.   uint8_t reply[8], idx = 0;   
  105.   while (true) {   
  106.     if (timer >= TIMEOUT) return ACK_TIMEOUT;   
  107.     while (!Serial1.available()) {   
  108.       delay(1);   
  109.       timer++;   
  110.       if (timer >= TIMEOUT) return ACK_TIMEOUT;   
  111.     }   
  112.     reply[idx] = Serial1.read();   
  113.     if ((idx == 0) && (reply[0] != CMD_HEAD)){   
  114.       timer++;   
  115.       delay(1);   
  116.       continue;   
  117.     }   
  118.     idx++;   
  119.     if (idx >= 8) {   
  120.       chk = 0;   
  121.       int i;   
  122.       for(i = 1; i <= 5; i++){   
  123.         chk ^= reply[i];   
  124.       }   
  125.       if(reply[1] != cmd || reply[6] != chk || reply[7] != CMD_TAIL){   
  126.         return ACK_FAIL;   
  127.       }   
  128.       for (i = 0; i < 3; i++) {   
  129.         packet[i] = reply[2+i];   
  130.       }   
  131.       return ACK_SUCCESS;   
  132.     }   
  133.   }   
  134.   return ACK_FAIL;   
  135. }
